usage: adaptive_runtime calibrate [-h] [--tasks-file TASKS_FILE]
                                  [--limit LIMIT] [--k K]
                                  [--avg-tokens AVG_TOKENS] [--fake] [--yes]
                                  [--keep] [--publish] [--offline]
                                  [--parallel PARALLEL]

options:
  -h, --help            show this help message and exit
  --tasks-file TASKS_FILE
                        任务清单路径
  --limit LIMIT         参与标定的任务数
  --k K                 每任务重复次数
  --avg-tokens AVG_TOKENS
                        单 run token 估算值（用于成本预估）
  --fake                使用受控模拟轨迹，不产生模型调用
  --yes                 跳过成本确认
  --keep                保留沙箱目录
  --publish             将结果写入 calibration.json
  --offline             离线模式（不联网，本地语料）
  --parallel PARALLEL   并发 run 数（默认 1=串行）
